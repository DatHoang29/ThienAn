**Architecture:** Backend uses Wolverine CQRS pattern (Controller → MessBus → Handler). Frontend adds a "Thiết bị" tab to the existing Zone management page with mapping grid, manual add, and auto-map preview.

**Tech Stack:** C# / SqlSugar / Wolverine (Backend), Vue 3 / TypeScript / Element Plus / VXE-Table (Frontend)

```csharp
using Mapster;
using Modules.TMS.Controllers.ZoneEquipment.Queries.Dto;
using Modules.TMS.Core.Entities;
using Modules.TMS.Infrastructure;
using Shared.DTO.Constants.Application;
using Shared.Infrastructure.Extensions;
using Shared.Infrastructure.Services;
using SqlSugar;
using Wolverine;

namespace Modules.TMS.Controllers.ZoneEquipment.Queries
{
    public class ZoneEquipmentQueryHandler : IWolverineHandler
    {
        private BaseRepository<TmsZoneEquipment> _zoneEquipmentRep;
        private BaseRepository<TmsZone> _zoneRep;
        private BaseRepository<TmsEquipment> _equipmentRep;
        private BaseCacheService _cache;

        public ZoneEquipmentQueryHandler(
            BaseRepository<TmsZoneEquipment> zoneEquipmentRep,
            BaseRepository<TmsZone> zoneRep,
            BaseRepository<TmsEquipment> equipmentRep,
            BaseCacheService cache)
        {
            _zoneEquipmentRep = zoneEquipmentRep;
            _zoneRep = zoneRep;
            _equipmentRep = equipmentRep;
            _cache = cache;
        }

        /// <summary>
        /// Description: Lấy danh sách ZoneEquipment theo bộ lọc
        /// </summary>
        public async Task<List<ZoneEquipmentOutput>> HandleAsync(ZoneEquipmentInput command)
        {
            var query = _zoneEquipmentRep.AsQueryable()
                .LeftJoin<TmsZone>((ze, z) => ze.ZoneId == z.ID)
                .LeftJoin<TmsEquipment>((ze, z, e) => ze.EquipmentId == e.ID)
                .WhereIF(!string.IsNullOrWhiteSpace(command.ZoneId), (ze, z, e) => ze.ZoneId == command.ZoneId)
                .WhereIF(!string.IsNullOrWhiteSpace(command.EquipmentId), (ze, z, e) => ze.EquipmentId == command.EquipmentId)
                .WhereIF(!string.IsNullOrWhiteSpace(command.Type), (ze, z, e) => ze.Type == command.Type)
                .WhereIF(command.Status != null, (ze, z, e) => ze.Status == command.Status)
                .WhereIF(!string.IsNullOrWhiteSpace(command.LaneId), (ze, z, e) => ze.LaneId == command.LaneId)
                .Where((ze, z, e) => ze.IsDelete == null)
                .OrderByDescending((ze, z, e) => ze.CreateTime)
                .Select((ze, z, e) => new ZoneEquipmentOutput
                {
                    // TmsZoneEquipment fields
                    ID = ze.ID,
                    Code = ze.Code,
                    ZoneId = ze.ZoneId,
                    EquipmentId = ze.EquipmentId,
                    Type = ze.Type,
                    Status = ze.Status,
                    LaneId = ze.LaneId,
                    Remark = ze.Remark,
                    CreateTime = ze.CreateTime,
                    // Joined fields
                    ZoneName = z.Name,
                    ZoneCode = z.Code,
                    EquipmentName = e.Name,
                    EquipmentCode = e.Code,
                    EquipmentKmNumber = e.KmNumber,
                    EquipmentMetNumber = e.MetNumber,
                });

            return await query.ToListAsync();
        }

        /// <summary>
        /// Description: Lấy danh sách phân trang ZoneEquipment
        /// </summary>
        public async Task<SqlSugarPagedList<ZoneEquipmentOutput>> HandleAsync(PageZoneEquipmentInput command)
        {
            var query = _zoneEquipmentRep.AsQueryable()
                .LeftJoin<TmsZone>((ze, z) => ze.ZoneId == z.ID)
                .LeftJoin<TmsEquipment>((ze, z, e) => ze.EquipmentId == e.ID)
                .WhereIF(!string.IsNullOrWhiteSpace(command.ZoneId), (ze, z, e) => ze.ZoneId == command.ZoneId)
                .WhereIF(!string.IsNullOrWhiteSpace(command.EquipmentId), (ze, z, e) => ze.EquipmentId == command.EquipmentId)
                .WhereIF(!string.IsNullOrWhiteSpace(command.Type), (ze, z, e) => ze.Type == command.Type)
                .WhereIF(!string.IsNullOrWhiteSpace(command.LaneId), (ze, z, e) => ze.LaneId == command.LaneId)
                .Where((ze, z, e) => ze.IsDelete == null)
                .OrderByDescending((ze, z, e) => ze.CreateTime)
                .Select((ze, z, e) => new ZoneEquipmentOutput
                {
                    ID = ze.ID,
                    Code = ze.Code,
                    ZoneId = ze.ZoneId,
                    EquipmentId = ze.EquipmentId,
                    Type = ze.Type,
                    Status = ze.Status,
                    LaneId = ze.LaneId,
                    Remark = ze.Remark,
                    CreateTime = ze.CreateTime,
                    ZoneName = z.Name,
                    ZoneCode = z.Code,
                    EquipmentName = e.Name,
                    EquipmentCode = e.Code,
                    EquipmentKmNumber = e.KmNumber,
                    EquipmentMetNumber = e.MetNumber,
                });

            return await query.ToPagedListAsync(command.Page ?? 1, command.PageSize ?? 50);
        }

        /// <summary>
        /// Description: Auto-map equipment cho một zone dựa trên km+met và LaneId
        /// Trả về danh sách preview, KHÔNG lưu DB
        /// </summary>
        public async Task<List<AutoMapPreviewOutput>> HandleAsync(AutoMapZoneEquipmentInput command)
        {
            var zone = await _zoneRep.GetFirstAsync(z => z.ID == command.ZoneId && z.IsDelete == null)
                ?? throw Oops.Oh(BaseMsg.BaseException.DataNotExist);

            return await GenerateAutoMapPreview(zone);
        }

        /// <summary>
        /// Description: Auto-map equipment cho tất cả zones
        /// Trả về danh sách preview, KHÔNG lưu DB
        /// </summary>
        public async Task<List<AutoMapPreviewOutput>> HandleAsync(AutoMapAllZoneEquipmentInput command)
        {
            var zones = await _zoneRep.AsQueryable()
                .Where(z => z.IsDelete == null && z.Status == BaseEnums.StatusEnum.Enable)
                .ToListAsync();

            var result = new List<AutoMapPreviewOutput>();
            foreach (var zone in zones)
            {
                var preview = await GenerateAutoMapPreview(zone);
                result.AddRange(preview);
            }
            return result;
        }

        /// <summary>
        /// Description: Sinh danh sách preview auto-map cho một zone
        /// Logic: So sánh vị trí km+met của equipment với ranh giới zone + khớp LaneId
        /// </summary>
        private async Task<List<AutoMapPreviewOutput>> GenerateAutoMapPreview(TmsZone zone)
        {
            var result = new List<AutoMapPreviewOutput>();

            // Tính vị trí ranh giới zone (đơn vị met)
            var zoneFromPos = (SqlFunc.IsNull(zone.FromKmNumber, 0) ?? 0) * 1000
                            + (SqlFunc.IsNull(zone.FromMetNumber, 0) ?? 0);
            var zoneToPos = (SqlFunc.IsNull(zone.ToKmNumber, 0) ?? 0) * 1000
                          + (SqlFunc.IsNull(zone.ToMetNumber, 0) ?? 0);

            // Bỏ qua zone không có thông tin vị trí
            if (zoneFromPos == 0 && zoneToPos == 0) return result;

            bool isGreater = zoneToPos >= zoneFromPos;

            // Lấy danh sách equipment đang hoạt động
            var allEquipment = await _equipmentRep.AsQueryable()
                .Where(e => e.IsDelete == null && e.Status == BaseEnums.StatusEnum.Enable)
                .ToListAsync();

            // Lấy danh sách equipment đã được mapped cho zone này
            var existingMappedIds = await _zoneEquipmentRep.AsQueryable()
                .Where(ze => ze.ZoneId == zone.ID && ze.IsDelete == null)
                .Select(ze => ze.EquipmentId)
                .ToListAsync();

            foreach (var equip in allEquipment)
            {
                // Bỏ qua equipment đã được mapped
                if (existingMappedIds.Contains(equip.ID)) continue;

                var equipPos = (equip.KmNumber ?? 0) * 1000 + (equip.MetNumber ?? 0);

                // Kiểm tra vị trí
                bool positionMatch;
                if (isGreater)
                    positionMatch = equipPos >= zoneFromPos && equipPos < zoneToPos;
                else
                    positionMatch = equipPos <= zoneFromPos && equipPos > zoneToPos;

                if (!positionMatch) continue;

                // Kiểm tra LaneId
                bool laneMatch = string.IsNullOrEmpty(zone.LaneId)
                    || string.IsNullOrEmpty(equip.LaneId)
                    || equip.LaneId == zone.LaneId;

                if (!laneMatch) continue;

                // Xây dựng lý do match
                var reasons = new List<string> { "Vị trí km+met" };
                if (!string.IsNullOrEmpty(zone.LaneId) && equip.LaneId == zone.LaneId)
                    reasons.Add("Lane trùng khớp");

                result.Add(new AutoMapPreviewOutput
                {
                    ZoneId = zone.ID,
                    ZoneName = zone.Name,
                    ZoneCode = zone.Code,
                    EquipmentId = equip.ID,
                    EquipmentName = equip.Name,
                    EquipmentCode = equip.Code,
                    KmNumber = equip.KmNumber,
                    MetNumber = equip.MetNumber,
                    LaneId = equip.LaneId,
                    MatchReason = string.Join(" + ", reasons)
                });
            }

            return result;
        }
    }
}
```

```csharp
using Mapster;
using Modules.TMS.Controllers.ZoneEquipment.Commands.Dto;
using Modules.TMS.Core.Entities;
using Modules.TMS.Infrastructure;
using Shared.DTO.Constants.Application;
using Shared.Infrastructure.Extensions;
using Shared.Infrastructure.Services;
using Wolverine;

namespace Modules.TMS.Controllers.ZoneEquipment.Commands
{
    public class ZoneEquipmentCommandHandler : IWolverineHandler
    {
        private BaseRepository<TmsZoneEquipment> _zoneEquipmentRep;
        private BaseCacheService _cache;

        public ZoneEquipmentCommandHandler(
            BaseRepository<TmsZoneEquipment> zoneEquipmentRep,
            BaseCacheService cache)
        {
            _zoneEquipmentRep = zoneEquipmentRep;
            _cache = cache;
        }

        /// <summary>
        /// Description: Thêm mới một ZoneEquipment mapping
        /// </summary>
        public async Task HandleAsync(AddZoneEquipmentInput command)
        {
            // Kiểm tra trùng lặp
            if (await _zoneEquipmentRep.IsAnyAsync(ze =>
                ze.ZoneId == command.ZoneId && ze.EquipmentId == command.EquipmentId && ze.IsDelete == null))
                throw Oops.Oh(BaseMsg.BaseException.Exist, "Equipment đã được mapping với Zone này");

            command.CreateUId = null;
            await _zoneEquipmentRep.InsertWithDiffLogAsync(command.Adapt<TmsZoneEquipment>());
            _cache.RemoveByPrefixKey(CacheConst.Tms.TmsZoneEquipment);
        }

        /// <summary>
        /// Description: Cập nhật một ZoneEquipment mapping
        /// </summary>
        public async Task HandleAsync(UpdateZoneEquipmentInput command)
        {
            var entity = await _zoneEquipmentRep.GetByIdAsync(command.ID)
                ?? throw Oops.Oh(BaseMsg.BaseException.DataNotExist);

            // Kiểm tra trùng lặp (trừ chính nó)
            if (await _zoneEquipmentRep.IsAnyAsync(ze =>
                ze.ZoneId == command.ZoneId && ze.EquipmentId == command.EquipmentId
                && ze.ID != command.ID && ze.IsDelete == null))
                throw Oops.Oh(BaseMsg.BaseException.Exist, "Equipment đã được mapping với Zone này");

            await _zoneEquipmentRep.UpdateWithDiffLogAsync(command.Adapt(entity));
            _cache.RemoveByPrefixKey(CacheConst.Tms.TmsZoneEquipment);
        }

        /// <summary>
        /// Description: Xóa một ZoneEquipment mapping
        /// </summary>
        public async Task HandleAsync(DeleteZoneEquipmentInput command)
        {
            var entity = await _zoneEquipmentRep.GetFirstAsync(ze => ze.ID == command.ID)
                ?? throw Oops.Oh(BaseMsg.BaseException.DataNotExist);

            await _zoneEquipmentRep.SoftDeleteAsync<TmsZoneEquipment>(entity);
            _cache.RemoveByPrefixKey(CacheConst.Tms.TmsZoneEquipment);
        }

        /// <summary>
        /// Description: Xóa nhiều ZoneEquipment mapping
        /// </summary>
        public async Task HandleAsync(List<DeleteZoneEquipmentInput> command)
        {
            var ids = command.Select(u => u.ID).ToList();

            var list = _zoneEquipmentRep.AsQueryable().Where(u => ids.Contains(u.ID));

            if (list?.Count() != ids.Count)
                throw Oops.Oh(BaseMsg.BaseException.DataNotMatch);

            await _zoneEquipmentRep.SoftDeleteAsync<TmsZoneEquipment>(l => ids.Contains(l.ID));
            _cache.RemoveByPrefixKey(CacheConst.Tms.TmsZoneEquipment);
        }

        /// <summary>
        /// Description: Xác nhận và lưu danh sách mapping từ auto-map preview
        /// </summary>
        public async Task HandleAsync(ConfirmZoneEquipmentMappingInput command)
        {
            // Nếu yêu cầu xóa mapping cũ cho zone cụ thể
            if (command.ClearExistingMappings && !string.IsNullOrWhiteSpace(command.ZoneId))
            {
                await _zoneEquipmentRep.SoftDeleteAsync<TmsZoneEquipment>(
                    ze => ze.ZoneId == command.ZoneId);
            }

            // Insert tất cả mapping mới
            foreach (var mapping in command.Mappings)
            {
                // Bỏ qua nếu đã tồn tại
                if (await _zoneEquipmentRep.IsAnyAsync(ze =>
                    ze.ZoneId == mapping.ZoneId && ze.EquipmentId == mapping.EquipmentId && ze.IsDelete == null))
                    continue;

                mapping.CreateUId = null;
                await _zoneEquipmentRep.InsertWithDiffLogAsync(mapping.Adapt<TmsZoneEquipment>());
            }

            _cache.RemoveByPrefixKey(CacheConst.Tms.TmsZoneEquipment);
        }
    }
}
```
